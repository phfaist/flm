% A page-width letterhead extends horizontally across the margins. Its height
% remains part of normal flow, and the body retains the configured text width.
\newcommand{\flmxLetterWideLogo}{%
  \noindent\makebox[\textwidth][l]{%
    \ifdefstring{\flmxLetterMode}{page-width}{\hspace*{-@@margin_left@@}}{}%
    \flmxLetterLogo}\par
}
\NewDocumentCommand{\flmxLetterHeader}{+m +m +m}{%
  \flmxLetterHeadfalse
  \ifflmxLetterHasLogo\ifflmxLetterColumns\else
    \flmxLetterWideLogo\flmxLetterHeadtrue
  \fi\fi
  \flmxLetterRowfalse
  \ifstrempty{#1}{}{\flmxLetterRowtrue}%
  \ifflmxLetterUS\ifstrempty{#3}{}{\flmxLetterRowtrue}\fi
  \ifflmxLetterHasLogo\ifflmxLetterColumns\flmxLetterRowtrue\fi\fi
  \ifflmxLetterRow
    \ifflmxLetterHead\flmxLetterGap{@@header_gap@@}\fi
    \noindent
    \ifflmxLetterUS
      \begin{minipage}[t]{\flmxLetterColumn}\vspace{0pt}%
        \ifflmxLetterHasLogo\ifflmxLetterColumns\flmxLetterLogo\fi\fi
      \end{minipage}\hfill
      \begin{minipage}[t]{\flmxLetterColumn}\vspace{0pt}\raggedright
        \ifstrempty{#1}{}{{\flmxLetterAddressFont\flmxLetterPlainField{#1}}}%
        \ifstrempty{#3}{}{\ifstrempty{#1}{}{\flmxLetterGap{5mm}}\flmxLetterPlainField{#3}}%
      \end{minipage}%
    \else
      \begin{minipage}[t]{\ifflmxLetterHasLogo\ifflmxLetterColumns\dimexpr\linewidth-@@logo_width@@-8mm\relax\else\linewidth\fi\else\linewidth\fi}%
        \vspace{0pt}\raggedright\flmxLetterAddressFont\flmxLetterPlainField{#1}%
      \end{minipage}%
      \ifflmxLetterHasLogo\ifflmxLetterColumns
        \hfill\begin{minipage}[t]{@@logo_width@@}\vspace{0pt}\raggedleft\flmxLetterLogo\end{minipage}%
      \fi\fi
    \fi\par
    \flmxLetterHeadtrue
  \fi
  \ifstrempty{#2}{}{\ifflmxLetterHead\flmxLetterGap{@@header_gap@@}\fi\flmxLetterAddress{#2}}%
  \ifflmxLetterUS
    \ifstrempty{#2}{\ifflmxLetterHead\flmxLetterGap{@@header_gap@@}\fi}{\flmxLetterGap{5mm}}%
  \else
    \ifstrempty{#3}{%
      \ifstrempty{#2}{\ifflmxLetterHead\flmxLetterGap{@@header_gap@@}\fi}{\flmxLetterGap{5mm}}%
    }{%
      \ifstrempty{#2}{\ifflmxLetterHead\flmxLetterGap{@@header_gap@@}\fi}{\flmxLetterGap{5mm}}%
      \flmxLetterRightLines{#3}\flmxLetterGap{7mm}%
    }%
  \fi
}
